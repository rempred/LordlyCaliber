cmake_minimum_required(VERSION 4.4)
project(LordlyCaliber VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Widgets Concurrent Test)
include(CTest)
add_library(lordly_engine STATIC engine/rom.cpp engine/resource.cpp engine/text.cpp engine/promotion.cpp engine/class_data.cpp engine/class_runtime_inventory.cpp engine/expanded_promotion_resource.cpp engine/save_runtime.cpp engine/project.cpp game/classes/save-format.c rebuild/compiler.cpp rebuild/class_runtime.cpp rebuild/class_menu_layout.cpp rebuild/promotion_runtime.cpp rebuild/save_runtime.cpp rebuild/elf_references.cpp)
target_include_directories(lordly_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(lordly_engine PUBLIC Qt6::Core)
qt_add_resources(lordly_engine game_sources PREFIX "/" BASE . FILES rebuild/toolchain.json game/accepted/func_000bbd50.c game/promotion/owners.json game/promotion/family-lookup.s game/promotion/family-menu.c game/promotion/class-selection.c game/promotion/class-discovery.c game/promotion/class_menu_config.h game/classes/class-runtime.c game/classes/relocations.json)
qt_add_resources(lordly_engine class_render_sources PREFIX "/" BASE . FILES game/promotion/class-render-rows.c game/promotion/class-render-entry.s game/promotion/class-render-style.s game/promotion/compatibility-mask.c)
qt_add_resources(lordly_engine equipment_menu_sources PREFIX "/" BASE . FILES game/promotion/equipment-class-menu.c game/promotion/class-categories.c game/promotion/category-class-counts.c game/promotion/class-screen.c game/promotion/equipment-first-class.s)
qt_add_resources(lordly_engine save_format_sources PREFIX "/" BASE . FILES game/classes/save-format.c game/classes/save_format.h game/classes/save-runtime.c game/classes/save-owners.json)
if(MSVC)
  target_compile_options(lordly_engine PUBLIC /W4 /permissive- /utf-8 /fp:strict)
endif()
qt_add_executable(LordlyCaliber app/main.cpp app/main_window.cpp app/record_model.cpp app/text_workspace.cpp app/promotion_editor.cpp)
target_link_libraries(LordlyCaliber PRIVATE lordly_engine Qt6::Widgets Qt6::Concurrent)
qt_add_resources(LordlyCaliber catalog PREFIX "/" BASE assets FILES assets/catalog.json)
set_target_properties(LordlyCaliber PROPERTIES WIN32_EXECUTABLE TRUE)
install(TARGETS LordlyCaliber RUNTIME DESTINATION .)
qt_add_executable(LordlyCaliber-cli app/cli.cpp)
target_link_libraries(LordlyCaliber-cli PRIVATE lordly_engine)
qt_add_resources(LordlyCaliber-cli catalog_cli PREFIX "/" BASE assets FILES assets/catalog.json)
set_target_properties(LordlyCaliber-cli PROPERTIES WIN32_EXECUTABLE FALSE)
install(TARGETS LordlyCaliber-cli RUNTIME DESTINATION .)
install(FILES ../LICENSE DESTINATION licenses RENAME LordlyCaliber.txt)
if(BUILD_TESTING)
  qt_add_executable(engine_tests tests/engine_tests.cpp)
  set_target_properties(engine_tests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(engine_tests PRIVATE lordly_engine Qt6::Test)
  add_test(NAME engine_contracts COMMAND engine_tests -o engine-test-report.txt,txt)
  qt_add_executable(ui_tests tests/ui_tests.cpp app/record_model.cpp app/main_window.cpp app/text_workspace.cpp app/promotion_editor.cpp)
  set_target_properties(ui_tests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(ui_tests PRIVATE lordly_engine Qt6::Widgets Qt6::Concurrent Qt6::Test)
  qt_add_resources(ui_tests catalog_test PREFIX "/" BASE assets FILES assets/catalog.json)
  add_test(NAME desktop_editing COMMAND ui_tests -platform offscreen -o ui-test-report.txt,txt)
  find_program(NODE_EXECUTABLE node)
  if(NODE_EXECUTABLE)
    add_test(NAME class_menu_mips COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/class_menu_mips.cjs")
    add_test(NAME original_save_codec COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/save-codec-mips.cjs")
  endif()
endif()
